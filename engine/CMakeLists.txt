cmake_minimum_required(VERSION 3.16)
project(Wake VERSION 0.1.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_WINDOWS_EXPORT_ALL_SYMBOLS ON)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
file(GLOB inventories CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/inventories/*.json")
set(generated_languages "${CMAKE_CURRENT_BINARY_DIR}/languages.generated.h")
add_custom_command(OUTPUT "${generated_languages}"
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/generate_languages.py"
          "${CMAKE_CURRENT_SOURCE_DIR}/inventories" "${generated_languages}"
  DEPENDS ${inventories} tools/generate_languages.py VERBATIM)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/sources.txt" runtime_sources)
add_library(kws_core OBJECT ${runtime_sources} "${generated_languages}")
if(MSVC)
  target_compile_options(kws_core PRIVATE /O2 /fp:strict)
else()
  target_compile_options(kws_core PRIVATE -O3 -ffp-contract=off -fno-fast-math)
endif()
target_include_directories(kws_core PRIVATE include ../include include/kws/detail src/session "${CMAKE_CURRENT_BINARY_DIR}")
add_library(kws_engine SHARED $<TARGET_OBJECTS:kws_core>)
add_library(Wake::engine ALIAS kws_engine)
target_include_directories(kws_engine PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
set_target_properties(kws_engine PROPERTIES VERSION 1.0.0 SOVERSION 1 EXPORT_NAME engine)
add_executable(kws-engine src/cli/main.cc)
target_link_libraries(kws-engine PRIVATE kws_engine)
target_include_directories(kws-engine PRIVATE include/kws/detail)
if(APPLE)
  set_target_properties(kws-engine PROPERTIES INSTALL_RPATH "@loader_path/../${CMAKE_INSTALL_LIBDIR}")
elseif(UNIX)
  set_target_properties(kws-engine PROPERTIES INSTALL_RPATH "$ORIGIN/../${CMAKE_INSTALL_LIBDIR}")
endif()
option(WAKE_BUILD_NODE "Build the optional Node-API addon" OFF)
if(WAKE_BUILD_NODE)
  find_path(NODE_API_INCLUDE node_api.h REQUIRED)
  add_library(wake_node MODULE ../sdk/nodejs/native/addon.cc $<TARGET_OBJECTS:kws_core>)
  target_include_directories(wake_node PRIVATE "${NODE_API_INCLUDE}" include)
  target_compile_definitions(wake_node PRIVATE NAPI_VERSION=8)
  set_target_properties(wake_node PROPERTIES PREFIX "" SUFFIX ".node")
  if(APPLE)
    target_link_options(wake_node PRIVATE -undefined dynamic_lookup)
  endif()
  if(WIN32)
    # Windows resolves napi_* through node.exe's import library (the file
    # node-gyp downloads from the same release directory).
    set(NODE_IMPORT_LIB "" CACHE FILEPATH "path to win-x64/node.lib for this node version")
    if(NOT EXISTS "${NODE_IMPORT_LIB}")
      message(FATAL_ERROR "WAKE_BUILD_NODE on Windows needs -DNODE_IMPORT_LIB=<path to win-x64/node.lib>")
    endif()
    target_link_libraries(wake_node PRIVATE "${NODE_IMPORT_LIB}")
  endif()
endif()
install(TARGETS kws_engine kws-engine EXPORT WakeTargets
  LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES ../include/phonlet_wake.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES include/kws/runtime.h include/kws/runtime.hpp include/kws/language.hpp DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/kws)
install(EXPORT WakeTargets NAMESPACE Wake:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Wake)
configure_package_config_file(cmake/WakeConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/WakeConfig.cmake" INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Wake)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/WakeConfigVersion.cmake" VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/WakeConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/WakeConfigVersion.cmake" DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Wake)
